# ─────────────────────────────────────────────────────────────────────────────
# BRING-Interface: Wochenplan in FHEM
#
# Einfügen über  set myCommander execute  (98_Commands.pm) – eine Zeile = ein
# Befehl, Zeilen mit # werden ignoriert.
#
# VORHER ANPASSEN (jeweils global suchen/ersetzen):
#   192.168.69.10:8095  ->  Adresse des BRING-Interface auf der NAS.
#                           ACHTUNG: der Port ist HOST_PORT aus der
#                           docker-compose.yml (Standard 8095) - NICHT 3000,
#                           das ist nur der Port INNERHALB des Containers.
#                           Auf 3000 antwortet auf der Synology oft ein
#                           anderer Dienst (z. B. Grafana) mit eigenem 401.
#                           Alternativ die Reverse-Proxy-Adresse, z. B.
#                           https://bring.<deine-domain>
#   DEINTOKEN           ->  Wert von API_TOKEN aus der .env des Containers
#                           (danach: docker compose up -d, nicht nur restart)
# ─────────────────────────────────────────────────────────────────────────────

# Abruf alle 5 Minuten.
define HTTP.Wochenplan HTTPMOD http://192.168.69.10:8095/api/fhem/plan?token=DEINTOKEN 300
attr HTTP.Wochenplan alias Wochenplan
attr HTTP.Wochenplan icon scene_dinner
attr HTTP.Wochenplan room Küche,FHEMVIZ->Küche
attr HTTP.Wochenplan group Küche
attr HTTP.Wochenplan enableControlSet 1
attr HTTP.Wochenplan timeout 15
attr HTTP.Wochenplan showMatched 0

# ── Readings aus der Antwort ─────────────────────────────────────────────────
# Absichtlich per Regex und NICHT per reading..JSON: bei der JSON-Auswertung
# dekodiert HTTPMOD Umlaute zu Perl-Zeichen, die FHEM dann falsch ausgibt
# ("Gef?llte Auberginen"). Die Regex nimmt die Bytes, wie sie kommen.
# Die Antwort ist flach und wird serverseitig von Anfuehrungszeichen befreit,
# deshalb genuegt "([^"]*)".

attr HTTP.Wochenplan reading01Name heute
attr HTTP.Wochenplan reading01Regex "today":"([^"]*)"
attr HTTP.Wochenplan reading02Name heute_status
attr HTTP.Wochenplan reading02Regex "today_status":"([^"]*)"
attr HTTP.Wochenplan reading03Name heute_bewertung
attr HTTP.Wochenplan reading03Regex "today_rating":"([^"]*)"
attr HTTP.Wochenplan reading04Name heute_sterne
attr HTTP.Wochenplan reading04Regex "today_stars":(\d+)
attr HTTP.Wochenplan reading05Name heute_zeit
attr HTTP.Wochenplan reading05Regex "today_time":"([^"]*)"
attr HTTP.Wochenplan reading06Name morgen
attr HTTP.Wochenplan reading06Regex "tomorrow":"([^"]*)"
attr HTTP.Wochenplan reading07Name woche
attr HTTP.Wochenplan reading07Regex "week":"([^"]*)"
attr HTTP.Wochenplan reading08Name geplant
attr HTTP.Wochenplan reading08Regex "planned":(\d+)
attr HTTP.Wochenplan reading09Name leere_tage
attr HTTP.Wochenplan reading09Regex "empty":(\d+)
attr HTTP.Wochenplan reading10Name stand

attr HTTP.Wochenplan reading12Regex "tue":"([^"]*)"

attr HTTP.Wochenplan reading18Name mo_sterne
attr HTTP.Wochenplan reading18Regex "mon_stars":(\d+)
attr HTTP.Wochenplan reading19Name di_sterne
attr HTTP.Wochenplan reading19Regex "tue_stars":(\d+)
attr HTTP.Wochenplan reading20Name mi_sterne

# ── Befehle ──────────────────────────────────────────────────────────────────
# Die Aktions-URLs antworten mit demselben JSON wie der Abruf, deshalb sind die
# Readings nach einem set sofort aktuell (setParseResponse 1).
attr HTTP.Wochenplan setParseResponse 1
attr HTTP.Wochenplan set01Name wuerfeln_woche
attr HTTP.Wochenplan set01URL http://192.168.69.10:8095/api/fhem/roll?scope=week&token=DEINTOKEN
attr HTTP.Wochenplan set01NoArg 1
attr HTTP.Wochenplan set02Name wuerfeln_heute
attr HTTP.Wochenplan set02URL http://192.168.69.10:8095/api/fhem/roll?scope=day&date=today&token=DEINTOKEN
attr HTTP.Wochenplan set02NoArg 1
attr HTTP.Wochenplan set03Name wuerfeln_morgen
attr HTTP.Wochenplan set03URL http://192.168.69.10:8095/api/fhem/roll?scope=day&date=tomorrow&token=DEINTOKEN
attr HTTP.Wochenplan set03NoArg 1
attr HTTP.Wochenplan set04Name wuerfeln_leere_tage
attr HTTP.Wochenplan set04URL http://192.168.69.10:8095/api/fhem/roll?scope=week&onlyEmpty=1&token=DEINTOKEN
attr HTTP.Wochenplan set04NoArg 1
attr HTTP.Wochenplan set05Name bewerten
attr HTTP.Wochenplan set05URL http://192.168.69.10:8095/api/fhem/rate?date=today&rating=$val&token=DEINTOKEN
attr HTTP.Wochenplan set05Hint lecker,gut,ok,maessig,schlecht,rausgeflogen,nie_wieder
attr HTTP.Wochenplan set06Name einkaufsliste
attr HTTP.Wochenplan set06URL http://192.168.69.10:8095/api/fhem/shopping?week=current&token=DEINTOKEN

# Vorlauf: was man am Vorabend noch anfangen muss (auftauen, einweichen).

attr HTTP.Wochenplan reading25Regex "today_prep":"([^"]*)"

attr HTTP.Wochenplan set07Name abgleichen

# ── Anzeige ──────────────────────────────────────────────────────────────────
attr HTTP.Wochenplan stateFormat Heute: heute
attr HTTP.Wochenplan webCmd wuerfeln_heute:wuerfeln_woche:bewerten
attr HTTP.Wochenplan webCmdLabel Heute würfeln:Woche würfeln:Bewerten
attr HTTP.Wochenplan event-on-change-reading .*

# ── FHEMVIZ ──────────────────────────────────────────────────────────────────
# Wochenplan als Kachel: Tage als Werte, Bewertung über den webCmd-Knopf.
attr HTTP.Wochenplan vizState heute
attr HTTP.Wochenplan vizReadings heute:Heute,morgen:Morgen,mo:Mo,di:Di,mi:Mi,do:Do,fr:Fr,sa:Sa,so:So
attr HTTP.Wochenplan sortby 5

# ── Optional: sonntags automatisch die neue Woche würfeln ────────────────────
define a_wochenplan_wuerfeln at *18:00:00 { if ((localtime)[6] == 0) { fhem("set HTTP.Wochenplan wuerfeln_woche") } }
attr a_wochenplan_wuerfeln alias Wochenplan sonntags würfeln
attr a_wochenplan_wuerfeln room Küche

# ── Optional: morgens einmal mit Mealie abgleichen ───────────────────────────
# Nur sinnvoll, wenn gleich darauf eine Ansage kommt - sonst reicht der
# automatische Abgleich der App.
define a_wochenplan_abgleich at *06:00:00 set HTTP.Wochenplan abgleichen
attr a_wochenplan_abgleich alias Wochenplan mit Mealie abgleichen
attr a_wochenplan_abgleich room Küche

# ── Optional: Aussentemperatur melden ────────────────────────────────────────
# Damit bevorzugt der Wuerfel bei Kaelte Eintopf und bei Hitze Salat. Der Wert
# gilt fuer heute und morgen; weiter voraus entscheidet der Monat.
# Quelle anpassen - hier der vorhandene Sensor Mobil3data.
define a_wochenplan_wetter at +*01:00:00 { GetFileFromURL("http://192.168.69.10:8095/api/fhem/weather?temp=".ReadingsVal("Mobil3data","temperature",15)."&token=DEINTOKEN") }
attr a_wochenplan_wetter alias Wochenplan Wetter melden
attr a_wochenplan_wetter room Küche

# ── Optional: abends an den Vorlauf erinnern ─────────────────────────────────
# Meldet sich nur, wenn morgen wirklich etwas vorzubereiten ist.
define a_wochenplan_vorbereitung at *20:00:00 { my $v = ReadingsVal("HTTP.Wochenplan","morgen_vorbereitung","");; my $m = ReadingsVal("HTTP.Wochenplan","morgen","");; if ($v ne "") { send_to_all("S", "Morgen gibt es $m - heute noch $v", "Wochenplan") } }
attr a_wochenplan_vorbereitung alias Wochenplan Vorbereitung
attr a_wochenplan_vorbereitung room Küche

# ── Optional: morgens ansagen, was es heute gibt ─────────────────────────────
# send_to_all ist die vorhandene Hausfunktion (siehe andere Notifys).
define a_wochenplan_ansage at *07:30:00 { my $e = ReadingsVal("HTTP.Wochenplan","heute","");; if ($e ne "") { send_to_all("S", "Heute gibt es: $e", "Wochenplan") } }
attr a_wochenplan_ansage alias Wochenplan Ansage
attr a_wochenplan_ansage room Küche
